chart.js scatter chart min-code

  • Sparisoma Viridi
5 Aug 2026 · 2 mins read · edit

It is a minimum code for displaying scatter chart using Chart.js library that is following window resizing.

Figure 1. An example of scatter chart with two series: ellipsoid (red) and infinity (blue);

The option to make both axex with the same scale has not yet been discovered. Futher modification will be reported in other post refering back to this one.

code

Code to get the result is as follows.

const cnt = document.getElementById("cnt");
Object.assign(cnt.style, {
  width: "100%",
});

const can = document.createElement("canvas");
Object.assign(can.style, {
  width: "100%",
  border: "1px solid var(--muted-border)",
});


function getThemeColor(variableName) {
  return getComputedStyle(document.documentElement)
    .getPropertyValue(variableName)
    .trim();
}


function generateData() {
  const n = 100;
  
  const s1 = [];
  for(let i = 0; i < n; i++) {
    const q = 2 * Math.PI * i / n;
    const x = Math.cos(q);
    const y = Math.sin(q);
    
    const p = {x: x, y: y};
    s1.push(p);
  }

  const s2 = [];
  for(let i = 0; i < n; i++) {
    const q = 2 * Math.PI * i / n;
    const x = Math.cos(q);
    const y = Math.sin(2*q);
    
    const p = {x: x, y: y};
    s2.push(p);
  }
  
  const data = {
    datasets: [{
      label: 'ellipsoid',
      data: s1,
      backgroundColor: 'rgb(255, 99, 132)'
    }, {
      label: 'infinity',
      data: s2,
      backgroundColor: 'rgb(99, 132, 255)'
    }],
  };
  
  return data;
}


const config = {
  type: 'scatter',
  data: generateData(),
  options: {
    responsive: false,
    maintainAspectRatio: false,
    scales: {
      x: {
        type: 'linear',
        position: 'bottom',
        grid: {
          color: getThemeColor('--muted-border'),
        },
      },
      y: {
        type: 'linear',
        position: 'left',
        grid: {
          color: getThemeColor('--muted-border'),
        },
      },
    },
  },
};

cnt.append(can);
new Chart(can, config);

To use the code in HTML a div element with id cnt must be preceeded the script element containing the code.

notes

  • The result is obtained using available examples 1 2 with modifications.
  • Border for canvas element is controlled via var(--muted-border) while for chart grid is via a helper fuction getThemeColor() 3.
