The code is as follows.

import { Tensor1D as T1D } from "endap"

let a = T1D.create(10);
a.fill(1);
let b = T1D.create(10, 1);
b[9] = -1;
let c = T1D.add(a, b);
let d = T1D.dot(a, b);

console.log("a =", a)
console.log("b =", b)
console.log("c =", c)
console.log("d =", d)

And

a = [1, 1, 1, 1, 1, 1, 1, 1, 1]
a = [1, 1, 1, 1, 1, 1, 1, 1, -1]
c = [2, 2, 2, 2, 2, 2, 2, 2, 0]
d = 8

are the results.
