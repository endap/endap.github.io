It is a post about fetching data in Markdown format from a localhost using simply Vanilla JS.

intro

The format is chosen since it can have metadata compared to CSV or TXT, but not too complex compared to JSON 1. There is a modern interface in JavaScript that allows you to make HTTP requests via the Fetch API, which replaces the older XMLHttpRequest method and provides a cleaner and more flexible way to fetch resources asynchronously 2. It will use the second last example from an available Medium story 3, which is modified to fit the purpose, especially how to call the function 4.

To assure that JS function split() works in known OS a regular expression [\r\n]+ is used 5. Markdown has to section, which are front matter and section 6. An in-house, simple TOML parser is used with some limitations 7. For the CSV it returns array of object instead of an array for each column 8.

There is away using parsing in datasets of Chart.js not to change the keys of object in the array 9. Number format must be modified for x-axis and also for the tooltips (the popups when mouseover event occurs over a point) to remove thousand commas 10.

async function loadData(url) {
  const response = await fetch(url);
  const md = response.text();
  return md;
}

function getTOMLfromMD(md) {
  const lines = md.split(/[\r\n]+/);
  const n = lines.length;
  let isMetadata = false;
  const temp = [];
  
  for(let i = 0; i < n; i++) {
    const l = lines[i];
    
    if(l.indexOf('+++') == 0 && isMetadata) {
      isMetadata = false;
      break;
    }
    
    if(l.indexOf('+++') == 0 && !isMetadata) {
      isMetadata = true;
    }
    
    if(isMetadata) {
      temp.push(l);
    }
  }
  
  temp.shift();
  const toml = temp.join('\n');
  return toml;  
}

function getValueFromTOML(toml, field) {
  const lines = md.split(/[\r\n]+/);
  const n = lines.length;
  let val = null;
  
  for(let i = 0; i < n; i++) {
    const l = lines[i];
    if(l.indexOf(field) == 0) {
      val = l
        .split('=')
        .slice(1)
        .join('=')
        .trim()
        .replace(/^['"]|['"]$/g, '');
      break;
    }
  }
  
  return val;
}

function getCSVfromMD(md) {
  const lines = md.split(/[\r\n]+/);
  const n = lines.length;
  let val = [];
  let isContent = false;
  
  for(let i = 0; i < n; i++) {
    const l = lines[i];
    
    if(isContent) {
      val.push(l);
    }
    
    if(l.indexOf('+++') == 0 && i > 0) {
      isContent = true;
    }
  }
  
  const csv = val.join('\n');
  return csv;
}

function getColumnsFromCSV(csv, sep) {
  const lines = csv.split(/[\r\n]+/);
  const n = lines.length;
  const keys = lines[0].split(sep);
  
  const data = [];
  for(let i = 1; i < n; i++) {
    const cols = lines[i].split(sep);
    const point = {};
    point[keys[0]] = cols[0];
    point[keys[1]] = cols[1];
    data.push(point);
  }
  
  return data;
}

const url = '/data/pop-id-1950-2026.md';
const md = await loadData(url);
const toml = getTOMLfromMD(md);
const csv = getCSVfromMD(md);

const sep = getValueFromTOML(toml, 'separator');
const cols = getColumnsFromCSV(csv, sep);
const keys = Object.keys(cols[0]);

const m = cols.length;
consoles.log(keys[0] + '\t' + keys[1]);
for(let i = 0; i < m; i++) {
  consoles.log(cols[i][keys[0]], '\t');
  consoles.log(cols[i][keys[1]], '\n');
}

const config = {

closing

Indonesian population data in Markdown format can be served under /data/ and obtained using fetch() function. The data is displayed in endap CLI and Chart.js scatter plot. The use of await might hide the error when a variable or function not defined, which can lead to unnoticed error.
