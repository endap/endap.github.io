sine curve

Following is illustration of a sine curve.

Following is code for above result.

/*
  data
*/

import { Tensor1D as T1D } from "endap"

const A = 0.1;
const T = 2;
const phi0 = 0;
const omega = 2 * Math.PI / T;

const tbeg = 0;
const tend = 4;
const dt = 0.1

const n = (tend - tbeg) / dt;
const x1 = T1D.create(n);
const x2 = T1D.create(n);

for(let i = 0; i < n; i++) {
  const t = tbeg + i * dt;
  const x = Math.sin(omega * t + phi0);
  x1[i] = t;
  x2[i] = x;
}

const data1 = T1D.create(n);

for(let i = 0; i < n; i++) {
  const point = {
    x: x1[i],
    y: x2[i],
  };
  data1[i] = point;
}

parabolic curve

Following is illustration of a parabolic curve.

Following is code for above result.

/*
  data
*/

import { Tensor1D as T1D } from "endap"

const x0 = 0;
const y0 = 0;
const v0x = 20;
const v0y = 40;
const ax = 0;
const ay = -10;

const tbeg = 0;
const tend = 8;
const dt = 0.5;

const n = (tend - tbeg) / dt;
const x1 = T1D.create(n);
const x2 = T1D.create(n);

for(let i = 0; i < n; i++) {
  const t = tbeg + i * dt;
  const x = x0 + v0x * t + 0.5 * ax * t * t;
  const y = y0 + v0y * t + 0.5 * ay * t * t;
  x1[i] = x;
  x2[i] = y;
}

const data1 = T1D.create(n);

for(let i = 0; i < n; i++) {
  const point = {
    x: x1[i],
    y: x2[i],
  };
  data1[i] = point;
}
