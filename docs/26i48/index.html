example-3

import { Console as CLI } from "endap"
const consoles = CLI.create("300px");

const cnt = document.getElementById("cnt3");
cnt.append(consoles.el);

function f(x) {
  let c0 = 21.3;
  let c1 = -10.3;
  let c2 = 1;
  let y = c0 + c1 * x + c2 * x * x;
  return y;
}


const xmin = 0;
const xmax = 10;
const dx = 1;
const n = (xmax - xmin) / dx;

for(let i = 0; i <= n; i++) {
  let x = xmin + i*dx;
  let y = f(x);
  
  consoles.log(x.toFixed(2), "\t");
  consoles.log(y.toFixed(2));
}
