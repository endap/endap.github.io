It is an advancement of previous post 1 with additional info, e.g. using object as dictionary 2.

for_0

Code

const x = [1, 3, 5, 7, 9];
const n = x.length;
for(let i = 0; i < n; i++) {
  console_log(x[i] + ", ");
}

Result

for_1

Code

const x = [1, 3, 5, 7, 9];
for(const i of x) {
  console_log(i + ", ");
}

Result

for_2

Code

const x = [1, 3, 5, 7, 9];
for(const [idx, val] of x.entries()) {
  console_log("x[" + idx + "] = " + val + "\n");
}

Result

for_3

Code

const x = {
  id: 2,
  name: "Albert Einstein",
  field: "physics",
};
for(const [key, val] of Object.entries(x)) {
  console_log(`${key}: ${val}` + "\n");
}

Result
