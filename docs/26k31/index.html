// Create canvas
const canvas = document.createElement("canvas");
canvas.width = 200;
canvas.height = 200;
canvas.style.border = "1px solid black";

// Create button
const button = document.createElement("button");
button.textContent = "start";

// Append elements to parent
cnt.appendChild(canvas);
cnt.appendChild(button);

// Canvas context
const ctx = canvas.getContext("2d");

// Circle properties
const circle = {
    x: 100,
    y: 50,
    radius: 10,
    dx: 2,
    dy: 2
};

let running = false;
let animationId;

// Draw circle
function draw() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);

    ctx.beginPath();
    ctx.arc(circle.x, circle.y, circle.radius, 0, Math.PI * 2);
    ctx.fillStyle = "blue";
    ctx.fill();
    ctx.closePath();
}

// Update circle position
function update() {
    circle.x += circle.dx;
    circle.y += circle.dy;

    // Bounce horizontally
    if (
        circle.x + circle.radius >= canvas.width ||
        circle.x - circle.radius <= 0
    ) {
        circle.dx *= -1;
    }

    // Bounce vertically
    if (
        circle.y + circle.radius >= canvas.height ||
        circle.y - circle.radius <= 0
    ) {
        circle.dy *= -1;
    }
}

// Animation loop
function animate() {
    if (!running) return;

    update();
    draw();

    animationId = requestAnimationFrame(animate);
}

// Toggle button
button.addEventListener("click", () => {
    running = !running;

    if (running) {
        button.textContent = "stop";
        animate();
    } else {
        button.textContent = "start";
        cancelAnimationFrame(animationId);
    }
});

// Initial drawing
draw();

function createWaveSimulator(cnt) {
    // ============================
    // 1. Create UI elements
    // ============================

    cnt.innerHTML = "";

    cnt.style.display = "flex";
    cnt.style.flexDirection = "column";
    cnt.style.gap = "10px";
    cnt.style.maxWidth = "800px";

    const canvas = document.createElement("canvas");
    canvas.width = 700;
    canvas.height = 300;
    canvas.style.width = "100%";
    canvas.style.border = "1px solid #ccc";
    canvas.style.background = "#fff";

    const textarea = document.createElement("textarea");
    textarea.rows = 5;
    textarea.style.width = "100%";
    textarea.style.boxSizing = "border-box";
    textarea.style.fontFamily = "monospace";

    textarea.value =
`amplitude = 50
period = 4
wavelength = 200
phase = 0`;

    const button = document.createElement("button");
    button.textContent = "start";
    button.style.width = "100px";
    button.style.padding = "8px";

    cnt.appendChild(canvas);
    cnt.appendChild(textarea);
    cnt.appendChild(button);

    const ctx = canvas.getContext("2d");

    // ============================
    // 2. Wave parameters
    // ============================

    let params = {
        amplitude: 50,
        period: 4,
        wavelength: 200,
        phase: 0
    };

    let running = false;
    let animationId = null;
    let startTime = 0;

    // ============================
    // 3. Read parameters
    // ============================

    function readParameters() {
        const lines = textarea.value.split("\n");

        const newParams = {};

        lines.forEach(line => {
            const parts = line.split("=");

            if (parts.length !== 2) return;

            const key = parts[0].trim().toLowerCase();
            const value = parseFloat(parts[1]);

            if (!Number.isFinite(value)) return;

            if (key === "amplitude")
                newParams.amplitude = value;

            if (key === "period")
                newParams.period = value;

            if (key === "wavelength")
                newParams.wavelength = value;

            if (key === "phase")
                newParams.phase = value;
        });

        params = {
            ...params,
            ...newParams
        };

        // Avoid division by zero
        params.period = Math.max(0.001, params.period);
        params.wavelength = Math.max(0.001, params.wavelength);
    }

    // ============================
    // 4. Draw coordinate system
    // ============================

    function drawAxes() {
        const w = canvas.width;
        const h = canvas.height;

        const centerY = h / 2;

        ctx.clearRect(0, 0, w, h);

        // Horizontal axis
        ctx.beginPath();
        ctx.strokeStyle = "#aaa";
        ctx.lineWidth = 1;
        ctx.moveTo(0, centerY);
        ctx.lineTo(w, centerY);
        ctx.stroke();

        // Vertical axis
        ctx.beginPath();
        ctx.moveTo(0, 0);
        ctx.lineTo(0, h);
        ctx.stroke();

        // Center line
        ctx.fillStyle = "#555";
        ctx.font = "12px Arial";
        ctx.fillText("y = 0", 5, centerY - 8);
    }

    // ============================
    // 5. Draw traveling wave
    // ============================

    function drawWave(time) {
        readParameters();

        drawAxes();

        const w = canvas.width;
        const h = canvas.height;

        const centerY = h / 2;

        const A = params.amplitude;
        const T = params.period;
        const lambda = params.wavelength;
        const phase = params.phase;

        const omega = 2 * Math.PI / T;
        const k = 2 * Math.PI / lambda;

        ctx.beginPath();
        ctx.strokeStyle = "#007bff";
        ctx.lineWidth = 2.5;

        for (let x = 0; x < w; x++) {

            // Traveling sinusoidal wave:
            // y(x,t) = A sin(kx - omega*t + phase)

            const y = A * Math.sin(
                k * x - omega * time + phase
            );

            const canvasY = centerY - y;

            if (x === 0) {
                ctx.moveTo(x, canvasY);
            } else {
                ctx.lineTo(x, canvasY);
            }
        }

        ctx.stroke();

        // Display information
        ctx.fillStyle = "#333";
        ctx.font = "13px Arial";

        ctx.fillText(
            `Amplitude: ${A} | Period: ${T} s | Wavelength: ${lambda} px`,
            10,
            20
        );
    }

    // ============================
    // 6. Animation loop
    // ============================

    function animate(timestamp) {
        if (!running) return;

        const elapsed = (timestamp - startTime) / 1000;

        drawWave(elapsed);

        animationId = requestAnimationFrame(animate);
    }

    // ============================
    // 7. Start / Stop button
    // ============================

    button.addEventListener("click", () => {

        if (!running) {

            // START
            running = true;
            button.textContent = "stop";

            readParameters();

            startTime = performance.now();

            animationId = requestAnimationFrame(animate);

        } else {

            // STOP
            running = false;
            button.textContent = "start";

            cancelAnimationFrame(animationId);
        }
    });

    // ============================
    // 8. Update parameters
    // ============================

    textarea.addEventListener("input", () => {
        if (!running) {
            drawWave(0);
        }
    });

    // Initial drawing
    drawWave(0);

    // Return public interface
    return {
        start: () => button.click(),
        stop: () => {
            if (running) button.click();
        },
        redraw: () => drawWave(0)
    };
}

// Parent container
// Assumes cnt is an existing div element.

cnt.innerHTML = "";

// Canvas
const canvas = document.createElement("canvas");
canvas.width = 600;
canvas.height = 400;
canvas.style.display = "block";
canvas.style.maxWidth = "100%";
canvas.style.border = "1px solid #ccc";
canvas.style.background = "#fafafa";

const ctx = canvas.getContext("2d");

// Toggle button
const button = document.createElement("button");
button.textContent = "start";
button.style.marginTop = "10px";
button.style.padding = "8px 24px";
button.style.cursor = "pointer";

// Add elements
cnt.appendChild(canvas);
cnt.appendChild(button);

// Pendulum parameters
const pivot = {
    x: canvas.width / 2,
    y: 50
};

const length = 220;
const maxAngle = Math.PI / 4; // 45 degrees
const gravity = 9.81;
const omega = Math.sqrt(gravity / (length / 100));

const bobRadius = 18;

// Simulation state
let running = false;
let animationId = null;

let startTime = null;
let elapsedTime = 0;
let periodCount = 0;

let previousAngle = -maxAngle;
let currentAngle = -maxAngle;

let lastPeriodAngle = -maxAngle;
let lastFrameTime = 0;

// Pendulum animation
function drawPendulum(angle, velocity) {

    ctx.clearRect(0, 0, canvas.width, canvas.height);

    // Pivot
    ctx.beginPath();
    ctx.arc(pivot.x, pivot.y, 5, 0, Math.PI * 2);
    ctx.fillStyle = "#333";
    ctx.fill();

    // Bob position
    const bobX = pivot.x + length * Math.sin(angle);
    const bobY = pivot.y + length * Math.cos(angle);

    // String
    ctx.beginPath();
    ctx.moveTo(pivot.x, pivot.y);
    ctx.lineTo(bobX, bobY);
    ctx.strokeStyle = "#555";
    ctx.lineWidth = 2;
    ctx.stroke();

    // Velocity-based color
    const normalizedVelocity = Math.min(
        1,
        Math.abs(velocity) / (omega * maxAngle)
    );

    // Blue (0 velocity) -> Red (maximum velocity)
    const red = Math.round(255 * normalizedVelocity);
    const blue = Math.round(255 * (1 - normalizedVelocity));

    const bobColor = `rgb(${red}, 40, ${blue})`;

    // Bob
    ctx.beginPath();
    ctx.arc(bobX, bobY, bobRadius, 0, Math.PI * 2);
    ctx.fillStyle = bobColor;
    ctx.fill();

    ctx.strokeStyle = "#333";
    ctx.lineWidth = 2;
    ctx.stroke();

    // Pivot label
    ctx.fillStyle = "#333";
    ctx.font = "14px Arial";
    ctx.fillText("Pivot", pivot.x + 12, pivot.y - 8);

    // Information
    ctx.fillStyle = "#222";
    ctx.font = "16px Arial";

    ctx.fillText(
        `Elapsed time: ${elapsedTime.toFixed(2)} s`,
        15, 300
    );

    ctx.fillText(
        `Completed periods: ${periodCount}`,
        15, 325
    );

    ctx.fillText(
        `Velocity: ${Math.abs(velocity).toFixed(2)} rad/s`,
        15, 350
    );
}

// Animation loop
function animate(timestamp) {

    if (!running) return;

    if (startTime === null) {
        startTime = timestamp;
        lastFrameTime = timestamp;
    }

    const dt = (timestamp - startTime) / 1000;

    // Small-angle harmonic oscillation
    // Starts at the leftmost position.
    currentAngle = maxAngle * Math.cos(omega * dt + Math.PI);

    // Angular velocity
    const velocity =
        -maxAngle * omega * Math.sin(omega * dt + Math.PI);

    // Elapsed time
    elapsedTime = dt;

    // Count complete periods
    const currentPhase = omega * dt;

    const newPeriodCount = Math.floor(
        currentPhase / (2 * Math.PI)
    );

    periodCount = newPeriodCount;

    drawPendulum(currentAngle, velocity);

    lastFrameTime = timestamp;

    animationId = requestAnimationFrame(animate);
}

// Start / Stop button
button.addEventListener("click", () => {

    if (!running) {

        running = true;
        button.textContent = "stop";

        // Continue from the current elapsed time
        startTime = null;

        const savedElapsed = elapsedTime;

        const originalAnimate = animate;

        function resumeAnimation(timestamp) {

            if (!running) return;

            if (startTime === null) {
                startTime = timestamp - savedElapsed * 1000;
            }

            originalAnimate(timestamp);
        }

        animationId = requestAnimationFrame(resumeAnimation);

    } else {

        running = false;
        button.textContent = "start";

        cancelAnimationFrame(animationId);
    }
});

// Initial drawing
drawPendulum(-maxAngle, 0);

function createWaterApp(cnt) {
    // Container styling
    cnt.style.display = "flex";
    cnt.style.alignItems = "center";
    cnt.style.justifyContent = "center";
    cnt.style.gap = "20px";
    cnt.style.padding = "20px";
    cnt.style.background = "#f1f5f9";
    cnt.style.borderRadius = "12px";

    // Canvas
    const canvas = document.createElement("canvas");
    canvas.width = 400;
    canvas.height = 350;
    canvas.style.maxWidth = "100%";

    const ctx = canvas.getContext("2d");

    // Vertical range slider
    const slider = document.createElement("input");
    slider.type = "range";
    slider.min = "10";
    slider.max = "90";
    slider.value = "50";

    slider.style.width = "180px";
    slider.style.height = "20px";
    slider.style.transform = "rotate(-90deg)";
    slider.style.cursor = "pointer";
    slider.style.accentColor = "#0284c7";

    cnt.appendChild(canvas);
    cnt.appendChild(slider);

    // Water container dimensions
    const container = {
        x: 70,
        y: 40,
        width: 260,
        height: 270
    };

    // Floating box dimensions
    const box = {
        width: 65,
        height: 45
    };

    function draw() {
        const waterLevel = Number(slider.value) / 100;

        ctx.clearRect(0, 0, canvas.width, canvas.height);

        const { x, y, width, height } = container;

        // Calculate water surface position
        const waterY = y + height * (1 - waterLevel);

        // Draw water
        ctx.save();

        ctx.beginPath();
        ctx.rect(x, y, width, height);
        ctx.clip();

        // Water body
        ctx.fillStyle = "#38bdf8";
        ctx.fillRect(
            x,
            waterY,
            width,
            y + height - waterY
        );

        // Water surface
        ctx.beginPath();
        ctx.moveTo(x, waterY);
        ctx.lineTo(x + width, waterY);
        ctx.strokeStyle = "#0284c7";
        ctx.lineWidth = 2;
        ctx.stroke();

        // Water surface highlight
        ctx.beginPath();
        ctx.moveTo(x + 10, waterY + 8);
        ctx.lineTo(x + width - 10, waterY + 8);
        ctx.strokeStyle = "rgba(255,255,255,0.35)";
        ctx.lineWidth = 2;
        ctx.stroke();

        ctx.restore();

        // Draw container outline
        ctx.beginPath();
        ctx.moveTo(x, y);
        ctx.lineTo(x, y + height);
        ctx.lineTo(x + width, y + height);
        ctx.lineTo(x + width, y);
        ctx.strokeStyle = "#334155";
        ctx.lineWidth = 5;
        ctx.lineJoin = "round";
        ctx.stroke();

        // Draw floating box
        const boxX = x + (width - box.width) / 2;
        const boxY = waterY - box.height;

        ctx.fillStyle = "#f59e0b";
        ctx.fillRect(
            boxX,
            boxY,
            box.width,
            box.height
        );

        // Box outline
        ctx.strokeStyle = "#92400e";
        ctx.lineWidth = 2;
        ctx.strokeRect(
            boxX,
            boxY,
            box.width,
            box.height
        );

        // Box label
        ctx.fillStyle = "#451a03";
        ctx.font = "bold 13px Arial";
        ctx.textAlign = "center";
        ctx.textBaseline = "middle";

        ctx.fillText(
            "BOX",
            boxX + box.width / 2,
            boxY + box.height / 2
        );

        // Water level label
        ctx.fillStyle = "#334155";
        ctx.font = "14px Arial";
        ctx.textAlign = "center";

        ctx.fillText(
            `Water Level: ${slider.value}%`,
            canvas.width / 2,
            330
        );
    }

    // Update canvas whenever slider changes
    slider.addEventListener("input", draw);

    // Initial rendering
    draw();
}

// Initialize app
createWaterApp(cnt);

    // =====================================
    // PHYSICS SIMULATION
    // =====================================

    function simulate(dt) {

        const waterY = getWaterY();
        const ballDensity = getBallDensity();

        // Ball mass is proportional to its density.
        // Volume is normalized to 1.
        const mass = ballDensity / waterDensity;

        // Determine submerged fraction.
        const submergedDepth =
            ball.y + radius - waterY;

        const submergedFraction = Math.max(
            0,
            Math.min(1, submergedDepth / (radius * 2))
        );

        // Gravitational acceleration
        const gravityAcceleration = gravity;

        // Buoyancy acceleration
        // Archimedes' principle:
        // Buoyant force = water density * g * displaced volume

        const buoyancyAcceleration =
            gravity *
            (waterDensity / ballDensity) *
            submergedFraction;

        // Viscous drag.
        // Drag opposes the direction of movement.
        // Drag increases with velocity.

        let dragAcceleration = 0;

        if (submergedFraction > 0) {

            dragAcceleration =
                -viscosity *
                submergedFraction *
                ball.vy /
                mass;
        }

        // Net acceleration
        const acceleration =
            gravityAcceleration -
            buoyancyAcceleration +
            dragAcceleration;

        // Integrate velocity
        ball.vy += acceleration * dt;

        // Integrate position
        ball.y += ball.vy * dt;

        // Aquarium bottom collision
        const bottom = tank.y + tank.height - radius;

        if (ball.y > bottom) {
            ball.y = bottom;
            ball.vy = 0;
        }

        // Aquarium ceiling collision
        const top = tank.y + radius;

        if (ball.y < top) {
            ball.y = top;
            ball.vy = 0;
        }

        // Stop when the ball reaches equilibrium.
        // This avoids unnecessary animation.
        if (
            Math.abs(ball.vy) < 0.1 &&
            Math.abs(acceleration) < 0.5
        ) {
            ball.vy = 0;
        }

        // Update information
        let status = "Falling";

        if (submergedFraction > 0 && ball.vy > 0) {
            status = "Sinking in water";
        }

        if (submergedFraction > 0 && ball.vy < 0) {
            status = "Rising due to buoyancy";
        }

        if (Math.abs(ball.vy) < 0.1) {
            status = "Equilibrium / Resting";
        }

        info.textContent =
            "Status: " + status +
            " | Velocity: " + ball.vy.toFixed(2) + " px/s" +
            " | Submerged: " +
            (submergedFraction * 100).toFixed(1) + "%";
    }

    // =====================================
    // ANIMATION LOOP
    // =====================================

    function animate(time) {

        if (!running) return;

        if (!previousTime) previousTime = time;

        let dt = (time - previousTime) / 1000;

        previousTime = time;

        // Prevent excessively large time steps
        dt = Math.min(dt, 0.03);

        // Substeps improve simulation stability
        const steps = 4;
        const subDt = dt / steps;

        for (let i = 0; i < steps; i++) {
            simulate(subDt);
        }

        drawAquarium();

        // Stop if the ball is stationary
        if (Math.abs(ball.vy) < 0.1) {

            const waterY = getWaterY();

            const submerged =
                ball.y + radius > waterY;

            const acceleration =
                gravity -
                gravity *
                (waterDensity / getBallDensity()) *
                Math.max(
                    0,
                    Math.min(
                        1,
                        (ball.y + radius - waterY) /
                        (radius * 2)
                    )
                );

            if (
                Math.abs(acceleration) < 0.5 ||
                ball.y >= tank.y + tank.height - radius
            ) {
                running = false;
                info.textContent = "Status: Simulation stopped";
                return;
            }
        }

        animationId = requestAnimationFrame(animate);
    }

    // Configuration
    const W = 600;
    const H = 500;

    const gravity = 980;       // px/s²
    const springK = 18;        // Spring stiffness
    const damping = 0.8;       // Air resistance
    const restLength = 150;    // Spring natural length
    const mass = 1;

    // Create SVG
    const svg = document.createElementNS(
        "http://www.w3.org/2000/svg", "svg"
    );

    svg.setAttribute("viewBox", `0 0 ${W} ${H}`);
    svg.style.width = "100%";
    svg.style.maxWidth = W + "px";
    svg.style.display = "block";
    svg.style.touchAction = "none";
    svg.style.background = "#f8fafc";

    cnt.appendChild(svg);

    const NS = "http://www.w3.org/2000/svg";

    function createElement(name, attrs) {
        const el = document.createElementNS(NS, name);

        for (const key in attrs) {
            el.setAttribute(key, attrs[key]);
        }

        svg.appendChild(el);
        return el;
    }

    // Fixed suspension point
    const anchor = {
        x: W / 2,
        y: 70
    };

    // Spring
    const spring = createElement("polyline", {
        fill: "none",
        stroke: "#64748b",
        "stroke-width": 3,
        "stroke-linecap": "round",
        "stroke-linejoin": "round"
    });

    // Bob
    const bob = createElement("circle", {
        cx: anchor.x,
        cy: anchor.y + restLength,
        r: 18,
        fill: "#2563eb",
        stroke: "#1e40af",
        "stroke-width": 2,
        cursor: "grab"
    });

    // Physics state
    const state = {
        x: anchor.x,
        y: anchor.y + restLength,
        vx: 0,
        vy: 0
    };

    let dragging = false;
    let lastTime = 0;

    // Generate spring geometry
    function updateSpring() {
        const dx = state.x - anchor.x;
        const dy = state.y - anchor.y;

        const length = Math.hypot(dx, dy);
        const ux = dx / length;
        const uy = dy / length;

        // Perpendicular direction
        const px = -uy;
        const py = ux;

        const coils = 14;
        const amplitude = 8;

        const points = [
            `${anchor.x},${anchor.y}`
        ];

        const segments = coils * 2;

        for (let i = 1; i < segments; i++) {
            const t = i / segments;

            const offset = (i % 2 === 0 ? -1 : 1) * amplitude;

            const x = anchor.x + ux * length * t + px * offset;
            const y = anchor.y + uy * length * t + py * offset;

            points.push(`${x},${y}`);
        }

        points.push(`${state.x},${state.y}`);

        spring.setAttribute("points", points.join(" "));
        bob.setAttribute("cx", state.x);
        bob.setAttribute("cy", state.y);
    }

    // Calculate acceleration using Hooke's law and gravity
    function physics(dt) {
        const dx = state.x - anchor.x;
        const dy = state.y - anchor.y;

        const length = Math.hypot(dx, dy);

        if (length < 0.001) return;

        // Spring force: F = -k * (length - restLength)
        const extension = length - restLength;

        const force = -springK * extension;

        const fx = force * dx / length;
        const fy = force * dy / length;

        // Gravity + spring force + damping
        const ax = fx / mass - damping * state.vx / mass;
        const ay = gravity + fy / mass
            - damping * state.vy / mass;

        // Semi-implicit Euler integration
        state.vx += ax * dt;
        state.vy += ay * dt;

        state.x += state.vx * dt;
        state.y += state.vy * dt;
    }

    // Convert pointer coordinates to SVG coordinates
    function pointerPosition(event) {
        const p = svg.createSVGPoint();

        p.x = event.clientX;
        p.y = event.clientY;

        return p.matrixTransform(
            svg.getScreenCTM().inverse()
        );
    }

    // Drag events
    bob.addEventListener("pointerdown", event => {
        dragging = true;

        bob.setPointerCapture(event.pointerId);
        bob.style.cursor = "grabbing";

        state.vx = 0;
        state.vy = 0;
    });

    bob.addEventListener("pointermove", event => {
        if (!dragging) return;

        const p = pointerPosition(event);

        state.x = p.x;
        state.y = p.y;

        state.vx = 0;
        state.vy = 0;

        updateSpring();
    });

    function releaseBob() {
        dragging = false;
        bob.style.cursor = "grab";
    }

    bob.addEventListener("pointerup", releaseBob);
    bob.addEventListener("pointercancel", releaseBob);

    // Animation loop
    function animate(time) {
        if (!lastTime) lastTime = time;

        // Limit timestep for numerical stability
        const dt = Math.min((time - lastTime) / 1000, 0.02);

        lastTime = time;

        if (!dragging) {
            physics(dt);
        }

        updateSpring();

        requestAnimationFrame(animate);
    }

    updateSpring();
    requestAnimationFrame(animate);

})();
