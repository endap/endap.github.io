In JS one way to perform loops is using for statement 1.

for_0

Code

for(let i = 0; i < 5; i++) {
  console_log(i + "\n");
}

Result

for_1

Code

let i = 0;
for(; i < 5; i++) {
  console_log(i + "\n");
}

Result

for_2

Code

let i = 0;
for(; ; i++) {
  if(i < 5) {} else break;
  console_log(i + "\n");
}

Result

for_3

Code

let i = 0;
for(; ;) {
  if(i < 5) {} else break;
  console_log(i + "\n");
  i++;
}

Result
