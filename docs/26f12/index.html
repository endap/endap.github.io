example

An example is given as follows.

Dimensions are 320px × 320px via canvas element container style, e.g. a div element.

functions

Following functions are defined to simplify creation of scatter plot but make it less flexible.

function defaultAxisTitle(text) {
  const axis = {
    title: {
      display: true,
      text: text,
      color: '#888',
      font: {
        size: 16,
        weight: 'bold'
      },
    },
    min: -1,
    max: 1,
    ticks: { stepSize: 0.4, color: "#888", },
    grid: { color: "rgba(128, 128, 128, 0.4)", },
    border: { color: "#888", width: 1, },
  };
  return axis;
}

function syntheticDataset1() {
  const x = [-0.6, -0.2, -0.1, 0.1, 0.2, 0.6];
  const y = [-0.6, -0.3, -0.1, 0.1, 0.3, 0.6]
  return {x: x, y: y};
}

function datasetToChartJS(dataset) {
  const nx = dataset.x.length;
  const ny = dataset.x.length;
  const n = Math.min(nx, ny);
  const xyValues = [];
  for(var i = 0; i < n; i++) {
    const xy = {
      x: dataset.x[i],
      y: dataset.y[i],
    };
    xyValues.push(xy);
  }
  return xyValues;
}

which will be used as follows.

const cnt = document.getElementById("container");
const can = document.createElement("canvas");

const data1 = syntheticDataset1();
const xyValues = datasetToChartJS(data1);

const dataset1 = {
  pointRadius: 4,
  pointBackgroundColor: "rgba(0, 0, 255, 0.8)",
  data: xyValues,
  showLine: true,
  borderWidth: 2,
  label: "dataset-1",
};

const xAxis = defaultAxisTitle("x");
const yAxis = defaultAxisTitle("y");

const chartOptions = {
  scales: {
    x: xAxis,
    y: yAxis,
  },
  responsive: true,            // Forces chart to fill the parent div
  maintainAspectRatio: false   // Unlocks custom width/height ratios
};

var chart = new Chart(can, {
  type: 'scatter',
  data: {
    datasets: [
      dataset1,
    ],
  },
  options: chartOptions,
});

cnt.style.width = "320px";
cnt.style.height = "320px";
cnt.append(can);

For other data, pairs of x[] and y[], new function, e.g. syntheticDataset2() must be defined first.
